  <script>
    let globalEntries = [];
    let currentSortColumn = -1;
    let currentSortAscending = true;

    // Helper: Parse DD-Mon-YY string (e.g., "15-Jul-26") to a Date object
    function parseDate(dateStr) {
      if (!dateStr) return new Date(0);
      const parts = dateStr.split('-');
      if (parts.length !== 3) return new Date(0);

      const day = parseInt(parts[0], 10);
      const months = {
        jan:0, feb:1, mar:2, apr:3, may:4, jun:5,
        jul:6, aug:7, sep:8, oct:9, nov:10, dec:11
      };
      const month = months[parts[1].toLowerCase()] || 0;

      let year = parseInt(parts[2], 10);
      if (year < 100) year += 2000;

      return new Date(year, month, day);
    }

    // Custom Helper: Formats "15-Jul-26" into "15.07.2026" (with leading zeros)
    function formatPrettyDate(dateStr) {
      const dateObj = parseDate(dateStr);
      if (isNaN(dateObj.getTime())) return dateStr; // fallback

      const day = String(dateObj.getDate()).padStart(2, '0');
      const month = String(dateObj.getMonth() + 1).padStart(2, '0'); // getMonth() is 0-indexed
      const year = dateObj.getFullYear();

      return `${day}.${month}.${year}`;
    }

    async function loadEntries() {
      try {
        const response = await fetch("data.csv");
        const text = await response.text();
        const rows = text.trim().split("\n").map(r => r.split(","));
        const data = rows.slice(1);

        let totalContrib = 0, totalExp = 0, balance = 0;
        globalEntries = [];

        data.forEach(row => {
          if (!row || row.length < 6) return;

          const type = row[3].toLowerCase();
          const amount = parseFloat(row[4]) || 0;
          balance = parseFloat(row[5]) || balance;
          if (type === "contribution") totalContrib += amount;
          if (type === "expense") totalExp += amount;

          globalEntries.push(row);
        });

        // Set top dashboard card values
        topContrib.textContent = totalContrib.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 });
        topExp.textContent = totalExp.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 });
        topBalance.textContent = balance.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 });

        // Keep bottom table summary matching
        summaryLine1.textContent = `Total Contributions: ${totalContrib}`;
        summaryLine2.textContent = `Total Expenses: ${totalExp}`;
        summaryLine3.textContent = `Current Balance: ${balance}`;

        // Reverse the items initially to show newest entries on top
        globalEntries.reverse();

        renderTable(globalEntries);
      } catch (err) {
        console.error("Failed to load entries:", err);
      }
    }

    function renderTable(entries) {
      const tbody = document.getElementById("entries");
      tbody.innerHTML = "";

      // Original CSV row index mapping:
      // [0] = Seq, [1] = Date, [2] = Person/Shop, [3] = Type, [4] = Amount, [5] = Balance, [6] = Description
      // Wanted layout: Date (1) -> Type (3) -> Amount (4) -> Balance (5) -> Person/Shop (2) -> Description (6)
      const renderOrder = [1, 3, 4, 5, 2, 6];

      entries.forEach(row => {
        const tr = document.createElement("tr");

        renderOrder.forEach(idx => {
          const cell = row[idx] || "";
          const td = document.createElement("td");

          if (idx === 1) {
            // Pretty custom dot format with leading zeros ("15.07.2026")
            td.textContent = formatPrettyDate(cell);
            td.classList.add("date");
          } else if (idx === 3) {
            // Pill Badge for Types
            const badge = document.createElement("span");
            badge.textContent = cell;
            badge.className = `badge ${cell.toLowerCase()}`;
            td.appendChild(badge);
          } else if (idx === 4) {
            // Colorized text for Amount
            const isExpense = row[3].toLowerCase() === 'expense';
            td.textContent = (isExpense ? "-$" : "$") + parseFloat(cell).toFixed(2);
            td.style.color = isExpense ? "#C53030" : "#2F855A";
            td.classList.add("amount");
          } else if (idx === 5) {
            // Formatted running balance
            const balanceVal = parseFloat(cell);
            td.textContent = (balanceVal < 0 ? "-$" : "$") + Math.abs(balanceVal).toFixed(2);
            td.style.fontWeight = "600";
          } else {
            td.textContent = cell;
          }

          if (idx === 6) {
            td.classList.add("description");
          }

          tr.appendChild(td);
        });
        tbody.appendChild(tr);
      });
    }

    function sortTable(colIndex, type) {
      if (currentSortColumn === colIndex) {
        currentSortAscending = !currentSortAscending;
      } else {
        currentSortColumn = colIndex;
        currentSortAscending = true;
      }

      globalEntries.sort((a, b) => {
        let valA = a[colIndex] ? a[colIndex].trim() : '';
        let valB = b[colIndex] ? b[colIndex].trim() : '';

        if (type === 'number') {
          return currentSortAscending
            ? (parseFloat(valA) || 0) - (parseFloat(valB) || 0)
            : (parseFloat(valB) || 0) - (parseFloat(valA) || 0);
        } else if (type === 'date') {
          const dateA = parseDate(valA);
          const dateB = parseDate(valB);
          return currentSortAscending ? dateA - dateB : dateB - dateA;
        } else {
          return currentSortAscending
            ? valA.localeCompare(valB)
            : valB.localeCompare(valA);
        }
      });

      // Update sort indicators (mapped to exact CSV indexes being sorted)
      const sortableIndexes = [1, 2, 3, 4, 5];
      sortableIndexes.forEach(i => {
        const iconSpan = document.getElementById(`sort-icon-${i}`);
        if (iconSpan) {
          if (i === colIndex) {
            iconSpan.textContent = currentSortAscending ? " ▲" : " ▼";
          } else {
            iconSpan.textContent = "";
          }
        }
      });

      renderTable(globalEntries);
    }

    const topContrib = document.getElementById("top-contrib");
    const topExp = document.getElementById("top-exp");
    const topBalance = document.getElementById("top-balance");
    const summaryLine1 = document.getElementById("summary-line1");
    const summaryLine2 = document.getElementById("summary-line2");
    const summaryLine3 = document.getElementById("summary-line3");

    loadEntries();
